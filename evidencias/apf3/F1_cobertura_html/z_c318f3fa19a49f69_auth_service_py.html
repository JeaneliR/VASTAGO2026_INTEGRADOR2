1"""Caso de uso de autenticación: login con bloqueo, refresh con rotación y logout."""

9# Hash ficticio para igualar el tiempo de respuesta cuando el usuario no existe (anti-enumeración) 

10import bcrypt as _bcrypt 

11_DUMMY = _bcrypt.hashpw(b"dummy-password", _bcrypt.gensalt(12)).decode()

14class AuthError(Exception): 

15 def __init__(self, msg, status=401): 

16 super().__init__(msg); self.status = status

19def login(email, password, ip, user_agent): 

20 """Los fallos se registran (contador + auditoría) y se confirman ANTES de lanzar el error, 

21 de lo contrario el rollback de la transacción borraría la evidencia del intento.""" 

22 cfg = current_app.config 

23 error = None 

24 with transaction() as cur: 

25 users, sess, aud = UsuarioRepository(cur), SesionRepository(cur), AuditoriaRepository(cur) 

26 u = users.find_by_email(email) 

27 if not u: 

28 sec.verify_password(password, _DUMMY) 

29 aud.registrar("LOGIN_FALLIDO", email=email, ip=ip, exito=False, detalle={"motivo": "usuario"}) 

30 error = AuthError("Credenciales inválidas") 

31 elif u["bloqueado_hasta"] and u["bloqueado_hasta"] > datetime.now(timezone.utc): 

32 aud.registrar("LOGIN_BLOQUEADO", u["id"], email, ip=ip, exito=False) 

33 error = AuthError("Cuenta bloqueada temporalmente por intentos fallidos", 423) 

34 elif not u["activo"] or not sec.verify_password(password, u["password_hash"]): 

35 r = users.registrar_fallo(u["id"], cfg["MAX_LOGIN_ATTEMPTS"], cfg["LOCKOUT_MINUTES"]) 

36 aud.registrar("LOGIN_FALLIDO", u["id"], email, ip=ip, exito=False, detalle={"intentos": r["intentos_fallidos"]}) 

37 error = AuthError("Credenciales inválidas") 

38 else: 

39 users.registrar_acceso(u["id"]) 

40 token, token_hash = sec.nuevo_refresh_token() 

41 sess.crear(u["id"], token_hash, datetime.now(timezone.utc) + timedelta(days=cfg["REFRESH_TOKEN_DAYS"]), ip, user_agent) 

42 aud.registrar("LOGIN_OK", u["id"], email, ip=ip) 

43 if error: 

44 raise error 

45 usuario = {"id": u["id"], "nombre": u["nombre"], "email": u["email"], "rol": u["rol"], "token_version": u["token_version"]} 

46 return usuario, sec.crear_access_token(usuario), token

49def refresh(refresh_token, ip, user_agent): 

50 """Rotación: el refresh token usado se revoca y se emite uno nuevo (detecta reutilización).""" 

51 cfg = current_app.config 

52 with transaction() as cur: 

53 users, sess, aud = UsuarioRepository(cur), SesionRepository(cur), AuditoriaRepository(cur) 

54 h = sec.hash_token(refresh_token) 

55 s = sess.buscar_valida(h) 

56 if not s: 

57 raise AuthError("Sesión inválida o expirada") 

58 u = users.get_con_rol(s["usuario_id"]) 

59 if not u or not u["activo"]: 

60 raise AuthError("Usuario inactivo") 

61 sess.revocar(h) 

62 nuevo, nuevo_hash = sec.nuevo_refresh_token() 

63 sess.crear(u["id"], nuevo_hash, datetime.now(timezone.utc) + timedelta(days=cfg["REFRESH_TOKEN_DAYS"]), ip, user_agent) 

64 aud.registrar("TOKEN_REFRESCADO", u["id"], u["email"], ip=ip) 

65 usuario = {"id": u["id"], "nombre": u["nombre"], "email": u["email"], "rol": u["rol"], "token_version": u["token_version"]} 

66 return usuario, sec.crear_access_token(usuario), nuevo
