32 t = unicodedata.normalize("NFKD", texto.lower()) 

33 t = "".join(c for c in t if not unicodedata.combining(c))

98def buscar_insumos(pregunta, insumos): 

99 """Devuelve (coincidencias, ambiguo). Pondera cada palabra por su rareza: 'suero' pesa más que 'leche'.""" 

100 m = re.findall(r"\b[a-z]{3}-[a-z]{3}-\d{2}\b", pregunta.lower()) 

101 if m: 

102 por_codigo = [i for i in insumos if i["codigo"].lower() in m] 

103 if por_codigo: 

104 return por_codigo[:MAX_INSUMOS], False 

105 q = set(tokens(pregunta)) 

106 if not q: 

107 return [], False 

108 toks = {i["codigo"]: set(tokens(i["nombre"])) for i in insumos} 

109 df = {} 

110 for ts in toks.values(): 

111 for t in ts: 

112 df[t] = df.get(t, 0) + 1 

113 puntaje = {c: sum(1 / df[t] for t in (q & ts)) for c, ts in toks.items()} 

114 mejor = max(puntaje.values(), default=0) 

115 if mejor <= 0: 

116 return [], False 

117 top = [i for i in insumos if abs(puntaje[i["codigo"]] - mejor) < 1e-9] 

118 return top[:6], len(top) > MAX_INSUMOS

121def parsear_periodo(t, corte): 

122 """Devuelve ('mes', mes_num | 'actual' | 'anterior', anio|None) o ('dias', n) o None.""" 

123 m = re.search(r"ultimos? (\d{1,3}) dias", t) 

124 if m: 

125 return ("dias", int(m.group(1))) 

126 if re.search(r"ultima semana|esta semana|ultimos dias", t): 

127 return ("dias", 7) 

128 if re.search(r"mes pasado|mes anterior", t): 

129 return ("mes", "anterior", None) 

130 if re.search(r"este mes|mes actual|mes en curso", t): 

131 return ("mes", "actual", None) 

132 anio = re.search(r"\b(20\d{2})\b", t) 

133 for palabra, num in MES_NUM.items(): 

134 if re.search(rf"\b{palabra}\b", t): 

135 return ("mes", num, int(anio.group(1)) if anio else None) 

136 return None

139def resolver_mes(periodo, corte, disponibles=None): 

140 """Convierte el periodo en 'YYYY-MM'. Sin año explícito, toma la ocurrencia más reciente (con datos si se indican).""" 

141 _, que, anio = periodo 

142 if que == "actual": 

143 return f"{corte.year}-{corte.month:02d}" 

144 if que == "anterior": 

145 y, m = (corte.year, corte.month - 1) if corte.month > 1 else (corte.year - 1, 12) 

146 return f"{y}-{m:02d}" 

147 if anio: 

148 return f"{anio}-{que:02d}" 

149 for y in (corte.year, corte.year - 1): 

150 ym = f"{y}-{que:02d}" 

151 if disponibles is None: 

152 if (y, que) <= (corte.year, corte.month): 

153 return ym 

154 elif ym in disponibles: 

155 return ym 

156 return None

159# ------------------------------------------------------------------ cálculo de cobertura (función pura, probada aparte) 

160def calcular_cobertura(insumo, base, reciente, dias_base, corte): 

161 """Días que alcanza el stock al ritmo de la ventana base y al ritmo de las últimas semanas. 

162 

163 base / reciente: filas de KardexRepository.consumo() para ese insumo (o None si no hubo salidas). 

164 """ 

165 stock, minimo = insumo["stock_actual"], insumo["stock_minimo"] 

166 tasa = (base["kg"] / dias_base) if base else 0.0 

167 tasa_rec = (reciente["kg"] / DIAS_RECIENTES) if reciente else 0.0 

168 out = {"stock": stock, "minimo": minimo, "tasa_base": tasa, "tasa_reciente": tasa_rec, 

169 "kg_base": base["kg"] if base else 0.0, "salidas_base": base["salidas"] if base else 0, 

170 "salida_tipica": base["mediana"] if base else None, "dias_base": dias_base, 

171 "dias": None, "dias_reciente": None, "agotamiento": None, "dias_a_minimo": None, "fecha_minimo": None, 

172 "bajo_minimo": stock < minimo} 

173 if tasa > 0: 

174 out["dias"] = int(stock / tasa) 

175 out["agotamiento"] = corte + dt.timedelta(days=out["dias"]) 

176 if stock > minimo: 

177 out["dias_a_minimo"] = int((stock - minimo) / tasa) 

178 out["fecha_minimo"] = corte + dt.timedelta(days=out["dias_a_minimo"]) 

179 if tasa_rec > 0: 

180 out["dias_reciente"] = int(stock / tasa_rec) 

181 return out

419def responder(pregunta): 

420 """Punto de entrada: texto libre -> respuesta estructurada. No lanza errores por preguntas que no entiende.""" 

421 t = norm(pregunta) 

422 if not t: 

423 return _resp_ayuda() 

424 kar, inicio, corte, insumos, dias_base, base, reciente = _contexto() 

425 intencion = detectar_intencion(t) 

426 periodo = parsear_periodo(t, corte) 

427 

428 if intencion == "ayuda": 

429 return _resp_ayuda() 

430 if intencion == "ranking": 

431 return _resp_ranking(insumos, dias_base, base, reciente, corte) 

432 if intencion == "bajo_minimo": 

433 return _resp_bajo_minimo(insumos, dias_base, base, reciente, corte) 

434 if intencion == "reproceso": 

435 return _resp_reproceso(t, corte, periodo) 

436 if intencion == "pronostico": 

437 return _resp_pronostico(t) 

438 if intencion == "produccion": 

439 return _resp_produccion(t, corte, periodo) 

440 

441 coinc, ambiguo = buscar_insumos(pregunta, insumos) 

442 if intencion is None and not coinc: 

443 return _resp("no_entendi", "No entendí esa pregunta. Puedo responder sobre insumos (stock, cuánto alcanza, consumo, ingresos), " 

444 "qué está bajo el mínimo y la producción de grageado. Prueba con alguna de estas:", sugerencias=SUGERENCIAS_BASE) 

445 if intencion is None and periodo and coinc: 

446 intencion = "consumo" 

447 if not coinc: 

448 return _resp(intencion, "¿De qué insumo? Dime el nombre (por ejemplo suero de leche, azúcar blanca, maní tostado entero) o su código, como LAC-SDL-02.", 

449 sugerencias=SUGERENCIAS_BASE[:3]) 

450 if ambiguo: 

451 return _pedir_precision(intencion or "stock", coinc, pregunta) 

452 

453 # Una sola respuesta por insumo (hasta MAX_INSUMOS); con varios, se unen los textos 

454 respuestas = [] 

455 for ins in coinc[:MAX_INSUMOS]: 

456 c = calcular_cobertura(ins, base.get(ins["codigo"]), reciente.get(ins["codigo"]), dias_base, corte) 

457 if intencion == "cobertura" or intencion is None: 

458 respuestas.append(_resp_cobertura(ins, c, corte, inicio)) 

459 elif intencion == "stock": 

460 respuestas.append(_resp_stock(ins, c, kar, corte)) 

461 elif intencion == "ingreso": 

462 respuestas.append(_resp_ingreso(ins, kar)) 

463 else: 

464 respuestas.append(_resp_consumo(ins, kar, corte, inicio, periodo, t)) 

465 if len(respuestas) == 1: 

466 return respuestas[0] 

467 unida = dict(respuestas[0]) 

468 unida["respuesta"] = "\n\n".join(r["respuesta"] for r in respuestas) 

469 unida["tarjetas"] = [] 

470 unida["tabla"] = None 

471 unida["nota"] = respuestas[0]["nota"] 

472 return unida
