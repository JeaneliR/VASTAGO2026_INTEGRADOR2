53def exportar_excel(data): 

54 from openpyxl import Workbook 

55 from openpyxl.styles import Font, PatternFill, Alignment 

56 from openpyxl.utils import get_column_letter 

57 book = Workbook() 

58 sheet = book.active 

59 sheet.title = "Reporte" 

60 sheet.append(["Vástago y Co",data["titulo"]]) 

61 sheet.append(["Generado",data["generado_en"]]) 

62 sheet.append(["Filtros",", ".join(f"{k}: {v}" for k,v in data["filtros"].items() if v) or "Sin filtros"]) 

63 sheet.append([]) 

64 sheet.append([c["titulo"] for c in data["columnas"]]) 

65 def safe(value): 

66 # El contenido del usuario se exporta como texto, nunca como fórmula. 

67 if isinstance(value,str) and value.startswith(("=","+","-","@","\t","\r")): 

68 return "'"+value 

69 return value 

70 for row in data["filas"]: 

71 sheet.append([safe(row.get(c["clave"])) for c in data["columnas"]]) 

72 sheet.freeze_panes="A6" 

73 sheet.auto_filter.ref=f"A5:{get_column_letter(len(data['columnas']))}{max(5,sheet.max_row)}" 

74 for cell in sheet[5]: 

75 cell.font=Font(bold=True,color="FFFFFF");cell.fill=PatternFill("solid",fgColor="4A2C1F") 

76 for index,column in enumerate(data["columnas"],1): 

77 letter=get_column_letter(index) 

78 sheet.column_dimensions[letter].width=34 if column["clave"] in ("insumo","producto","responsable") else 19 

79 sheet.column_dimensions['B'].width=42 

80 stream=BytesIO();book.save(stream);stream.seek(0) 

81 return stream
