130class TrazabilidadRepository(BaseRepository): 

131 """Consultas de genealogía: hacia atrás (de PT a lotes de insumo) y hacia adelante (de lote de insumo a PT y destinos)."""

134 def buscar(self, q): 

135 patron = "%" + q.replace("\\", "\\\\").replace("%", "\\%").replace("_", "\\_") + "%" 

136 pt = self._run("SELECT l.codigo, 'PT' AS tipo, p.nombre AS descripcion FROM lotes_pt l JOIN productos p ON p.id = l.producto_id " 

137 "WHERE l.codigo ILIKE %s ORDER BY l.codigo LIMIT 15", (patron,)) 

138 ins = self._run("SELECT l.codigo, 'INSUMO' AS tipo, i.nombre AS descripcion FROM lotes_insumo l JOIN insumos i ON i.id = l.insumo_id " 

139 "WHERE l.codigo ILIKE %s ORDER BY l.codigo LIMIT 15", (patron,)) 

140 return pt + ins

150 def etapas_con_consumos(self, orden_id): 

151 etapas = self._run( 

152 "SELECT e.id, e.secuencia, e.nombre, e.horas_estandar::float AS \"horasEstandar\", u.nombre AS responsable, q.codigo AS equipo, " 

153 "e.inicio, e.fin, CASE WHEN e.fin IS NOT NULL THEN round((extract(epoch FROM (e.fin - e.inicio)) / 3600)::numeric, 1)::float END AS \"horasReales\", " 

154 "e.merma_kg::float AS merma, e.observaciones " 

155 "FROM orden_etapas e LEFT JOIN usuarios u ON u.id = e.responsable_id LEFT JOIN equipos q ON q.id = e.equipo_id " 

156 "WHERE e.orden_id = %s ORDER BY e.secuencia", (orden_id,)) 

157 cons = self._run( 

158 "SELECT c.orden_etapa_id, i.nombre AS insumo, i.unidad, l.codigo AS lote, l.proveedor, " 

159 "to_char(l.fecha_vencimiento,'YYYY-MM-DD') AS vencimiento, c.cantidad::float AS cantidad " 

160 "FROM etapa_consumos c JOIN orden_etapas e ON e.id = c.orden_etapa_id JOIN lotes_insumo l ON l.id = c.lote_insumo_id " 

161 "JOIN insumos i ON i.id = l.insumo_id WHERE e.orden_id = %s ORDER BY c.id", (orden_id,)) 

162 for e in etapas: 

163 e["consumos"] = [{k: v for k, v in c.items() if k != "orden_etapa_id"} for c in cons if c["orden_etapa_id"] == e["id"]] 

164 return etapas
